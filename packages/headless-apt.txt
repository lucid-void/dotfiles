# Headless package list — Debian/Ubuntu equivalent of headless.txt, installed via
# apt-get on any host with apt-get instead of pacman (VMs, Codespaces).
#
# One package per line, `#` starts a comment. Installed by install.sh with
# `apt-get install -y --no-install-recommends`, so re-runs are cheap.
#
# Not a 1:1 name mapping for everything — a few headless.txt tools have no apt
# package at all (starship, rustup, yq, lazygit, gdu, bottom, fastfetch,
# carapace, gh), and two — neovim and rclone — have one that is too old to do
# the job, so all of them are installed separately by install.sh via official
# installers, GitHub release binaries, or a third-party apt repo. See install.sh's
# "Non-Arch (apt) package install" section for those.

# ── Toolchain ──────────────────────────────────────────────────
build-essential
git

# ── Fetching and archives ──────────────────────────────────────
curl
wget
tar
unzip
zip
rsync
# rclone is deliberately absent, like neovim below: trixie ships 1.60.1 and the
# Filen backend needs 1.73.0, so `rclone config` would not offer filen at all.
# install.sh's install_rclone fetches the upstream zip into ~/.local/bin.

# ── Shell and terminal ─────────────────────────────────────────
zsh
tmux
less

# ── Remote access ──────────────────────────────────────────────
openssh-client

# ── Docs ───────────────────────────────────────────────────────
man-db
manpages

# ── Editor ─────────────────────────────────────────────────────
# neovim is deliberately absent: trixie ships 0.10.4 and Ubuntu 24.04 ships
# 0.9.5, and LazyVim requires 0.11.2 or it refuses to start. install.sh's
# install_neovim fetches the upstream tarball into ~/.local instead. Listing it
# here as well would just install a second, older nvim at /usr/bin/nvim that
# ~/.local/bin shadows anyway.

# ── Shell tooling ──────────────────────────────────────────────
# Mirrors the same section in headless.txt. Without these, scripts/lint.sh
# aborts on its first check ("shellcheck is not installed") on every Debian and
# Ubuntu host — so the repo could not be linted on exactly the machines the apt
# path exists to provision.
shellcheck
shfmt

# ── Git ────────────────────────────────────────────────────────
git-delta

# ── Modern CLI replacements ────────────────────────────────────
# bat and fd-find install their binaries as `batcat`/`fdfind` on Debian/Ubuntu —
# install.sh symlinks them to `bat`/`fd` in ~/.local/bin after this list installs.
eza
bat
fd-find
ripgrep
fzf
zoxide

# ── CLI utilities ──────────────────────────────────────────────
jq

# ── System monitors ────────────────────────────────────────────
btop

# ── Languages ──────────────────────────────────────────────────
python3
python3-venv
pipx
nodejs
npm
