# Desktop package list — installed only when install.sh runs in desktop mode,
# on top of headless.txt.
#
# Official Arch repos only. One package per line; `#` starts a comment.
# AUR desktop apps live in desktop-aur.txt.

# ── Fonts ──────────────────────────────────────────────────────
# fontconfig provides fc-cache, which run_once_install_fonts.sh calls after
# dropping MesloLGS NF into ~/.local/share/fonts. That font is fetched by
# that script, so it is deliberately not here.
# Noto covers everything the Nerd Font doesn't: CJK glyphs and colour emoji.
fontconfig
noto-fonts
noto-fonts-emoji

# ── Terminals ──────────────────────────────────────────────────
# All three are configured under dot_config, kept in step with each other
# (same font, theme, padding, scrollback, copy-on-select, opacity). Ghostty
# left the AUR and lives in extra as of 1.3.x, so it belongs here rather than
# in desktop-aur.txt.
alacritty
ghostty
kitty

# ── Desktop integration ────────────────────────────────────────
# xdg-open, plus update-desktop-database for the browser entries installed by
# dot_local/share/applications/{brave-origin,chromium}.desktop.tmpl.
# desktop-file-utils also provides desktop-file-validate for checking them.
xdg-utils
desktop-file-utils

# ── Clipboard ──────────────────────────────────────────────────
# Wayland clipboard bridge for nvim and tmux. Swap for xclip on X11.
wl-clipboard

# ── Downloads ──────────────────────────────────────────────────
# Here for the browser, not for the CLI, which is why it is desktop-only:
# Chromium-family browsers download on exactly one TCP connection and expose no
# setting to change it. The Aria2 Explorer extension in browser-extensions.txt
# cancels Brave's download and re-issues it to the aria2 daemon over JSON-RPC.
# dots/run_setup_aria2.sh.tmpl owns the daemon; it no-ops until this is installed.
aria2

# ── Chat ───────────────────────────────────────────────────────
signal-desktop
telegram-desktop

# ── Hardware and CAD ───────────────────────────────────────────
# solaar manages Logitech Unifying receivers; it installs udev rules, so a
# reboot or `udevadm control --reload` is needed before it sees a device.
kicad
solaar

# ── Games ──────────────────────────────────────────────────────
prismlauncher
